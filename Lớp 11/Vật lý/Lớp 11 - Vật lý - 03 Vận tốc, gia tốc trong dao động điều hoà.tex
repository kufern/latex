	\section{Hệ thức độc lập thời gian}
		\subsection{Giữa vận tốc và li độ}
			\vspace{-5pt}
			\begin{align*}
				\text{Tại thời điểm } t \text{ có} & \:
				\begin{cases}
					x = A \cos \left( \omega t + \varphi \right) \\
					v = -\omega A \hspace{1.5pt} \sin \left( \omega t + \varphi \right)
				\end{cases} \\[3pt]
				\Leftrightarrow & \:
				\begin{cases}
					\dfrac{x}{\rule[0pt]{0pt}{11pt} A} = \cos \left( \omega t + \varphi \right) \\[10pt]
					\dfrac{v}{\rule[0pt]{0pt}{11pt} -\omega A} = \sin \left( \omega t + \varphi \right)
				\end{cases} \\[3pt]
				\Rightarrow & \enspace \frac{x^2}{\rule[0pt]{0pt}{12pt} A^2} + \frac{v^2}{\rule[0pt]{0pt}{12pt} \omega^2 A^2} = \cos^2 \left( \omega t + \varphi \right) + \sin^2 \left( \omega t + \varphi \right) = 1
			\end{align*}

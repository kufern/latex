	\section{Độ lệch pha}
		\subsection{Độ lệch pha của một dao động tại hai thời điểm}
			\vspace{5pt}
			\[\begin{aligned}
				\Delta \phi &= \phi_2 - \phi_1 \\
				&= \left( \omega \hspace{0.25pt} t_2 + \varphi \right) - \left( \omega \hspace{0.25pt} t_1 + \varphi \right) \\
				&= \omega \hspace{0.25pt} t_2 - \omega \hspace{0.25pt} t_1 \\
				&= \omega \left( t_2 - t_1 \right) \\
				&= \omega \times \Delta t \\[0.05cm]
				&\Rightarrow \; \custombox[2pt][1pt]{\Delta \phi = \omega \times \Delta t \hspace{1pt}}
			\end{aligned}\]

		\subsection{Độ lệch pha của hai dao động cùng phương, cùng tần số ở một thời điểm}
			\vspace{5pt}
			\[\begin{aligned}
				\Delta \phi &= \phi_2 - \phi_1 \\
				&= \left( \omega \hspace{0.25pt} t + \varphi_2 \right) - \left( \omega \hspace{0.25pt} t + \varphi_1 \right) \\
				&= \varphi_2 - \varphi_1 \\[0.08cm]
				&\Rightarrow \custombox[2pt][1pt]{\Delta \phi = \varphi_2 - \varphi_1}
			\end{aligned}\]
			
			\begin{center}
				\vspace{3pt}
				\textcolor{red}{\underline{\textbf{Chú ý:}}\hspace{5pt}
					\(\begin{aligned}[t]
						\varphi_1 &> \varphi_2 \Rightarrow \text{Dao động 1 sớm pha hơn dao động 2} \\
						\varphi_1 &< \varphi_2 \Rightarrow \text{Dao động 1 trễ pha hơn dao động 2} \\
						\varphi_1 &= \varphi_2 \Rightarrow \text{Hai dao động cùng pha (đồng pha)} \\
						\varphi_1 &= \varphi_2 \hspace{0.5pt} \pm \hspace{0.5pt} \pi \Rightarrow \text{Dao động 1 ngược pha với dao động 2} \\
					\end{aligned}\)}
			\end{center}

	\section{Động năng}
		\vspace{5pt}
		\begin{empheq}[box={\custombox[2pt][2pt]}]{equation*}
			W_\text{đ} = \frac{1}{2} \hspace{1pt} m \hspace{1pt} \omega^2 \hspace{-1pt} \left( A^2 - x^2 \right)
		\end{empheq}
		\begin{empheq}[box={\custombox[2pt][2pt]}]{equation*}
			W_\text{đ} = \frac{1}{2} \hspace{1pt} m \hspace{1pt} \omega^2 A^2 \sin^2 \left( \omega t + \varphi \right)
		\end{empheq} \\ [9pt]
		\underline{\textbf{\large Nhận xét:}}
		\vspace{5pt}
		\begin{itemize}
			\item \(W_\text{đ}\) biến thiên theo \(t\), với \( \hspace{1pt} \omega_\text{\hspace{1pt}đ} = 2 \hspace{1pt} \omega \) và \( \hspace{1pt} T_\text{\hspace{1pt}đ} = \dfrac{T}{2} \)
			\item Giá trị: \( 0 \leq W_\text{đ} \leq \dfrac{1}{2} \hspace{1pt} m \hspace{1pt} \omega^2 A^2 \)
		\end{itemize}

	\vspace{9pt}
	\section{Thế năng}

		\end{empheq}
		\begin{empheq}[box={\custombox[2pt][2pt]}]{equation*}
			W_\text{t} = \frac{1}{2} \hspace{1pt} m \hspace{1pt} \omega^2 A^2 \cos^2 \left( \omega t + \varphi \right)
		\end{empheq} \\ [9pt]
		\underline{\textbf{\large Nhận xét:}}
		\vspace{5pt}

			\item Giá trị: \( 0 \leq W_\text{t} \leq \dfrac{1}{2} \hspace{1pt} m \hspace{1pt} \omega^2 A^2 \)
		\end{itemize}

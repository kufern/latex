	\subsection{Nhận xét}
		\vspace{5pt}
		\begin{itemize}
			\item \(v\) biến thiên điều hoà theo \(t\), với tần số góc là \(\,\omega\)
			\item \(v\) sớm pha hơn \(x\) một góc \( \, \dfrac{\pi}{2}\) \: (\( v \perp x \))
			\item Giá trị vận tốc: \quad \( -\omega A \leq v \leq \omega A \)
				\begin{itemize}[label=\textopenbullet]
					\item \( v_\text{max} = \omega A \) khi vật qua VTCB theo chiều dương (\(+\))
					\item \( v_\text{min} = - \omega A \) khi vật qua VTCB theo chiều âm (\(-\))
				\end{itemize}
			\item Độ lớn vận tốc (Tốc độ): \quad \( 0 \leq \lvert v \rvert \leq \omega A \)
				\begin{itemize}[label=\textopenbullet]
					\item \( {\lvert v \rvert}_\text{max} = \omega A \) khi vật qua VTCB
					\item \( {\lvert v \rvert}_\text{min} = 0 \) khi vật qua VTB
				\end{itemize}
		\end{itemize}

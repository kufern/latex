\def\@custombox[#1]{%
	\@ifnextchar[%]
		{\@@custombox[#1]}%
		{\@@custombox[#1][0pt]}}
\def\@@custombox[#1][#2]#3{%
	\sbox\tempbox{#3}%
	\templength\ht\tempbox
	\advance\templength #1\relax
	\ht\tempbox\templength
	\templength\dp\tempbox
	\advance\templength #2\relax
	\dp\tempbox\templength
	\fbox{\usebox{\tempbox}}}%
\makeatother
